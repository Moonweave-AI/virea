cmake_minimum_required(VERSION 3.20)
project(virea_pose LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The VIREA SteamVR driver requires Windows x64")
endif()
if(NOT EXISTS "${OPENVR_INCLUDE_DIR}/openvr_driver.h")
  message(FATAL_ERROR "Set OPENVR_INCLUDE_DIR to the pinned official OpenVR headers")
endif()
add_library(driver_virea_pose SHARED driver.cpp)
option(VIREA_DESKTOP_COMPOSITOR "Use SteamVR's SimpleHMD desktop compositor path" ON)
target_compile_features(driver_virea_pose PRIVATE cxx_std_17)
target_include_directories(driver_virea_pose PRIVATE "${OPENVR_INCLUDE_DIR}")
target_compile_definitions(driver_virea_pose PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_definitions(driver_virea_pose PRIVATE
  VIREA_DESKTOP_COMPOSITOR=$<BOOL:${VIREA_DESKTOP_COMPOSITOR}>)
target_link_libraries(driver_virea_pose PRIVATE ws2_32 dxgi d3d11)
set_target_properties(driver_virea_pose PROPERTIES
  RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/driver/bin/win64")
file(COPY resources DESTINATION "${CMAKE_BINARY_DIR}/driver")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/driver.vrdrivermanifest" driver_manifest)
if(VIREA_DESKTOP_COMPOSITOR)
  string(JSON driver_manifest SET "${driver_manifest}" redirectsDisplay false)
endif()
file(WRITE "${CMAKE_BINARY_DIR}/driver/driver.vrdrivermanifest" "${driver_manifest}\n")
